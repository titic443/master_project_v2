# PICT Constraints for Job Search Page
# =====================================
# อ้างอิงข้อมูลจริงใน DB:
#   title: "Software Engineer"   category: "Engineering"
#   employmentType: "Freelance"  location: "Bangkok, Thailand"
#   salaryMin: 30000  salaryMax: 70000  isRemote: true
#
# โครงสร้างไฟล์:
#   - Format A (Dataset Override) : `key.valid = ...` / `key.invalid = ...`
#     ระบบ Dart จะ override ค่าใน datasets.json ก่อนส่งไป PICT
#   - Format B (PICT IF/THEN)     : `IF [k] = "v" THEN [k2] = "v2";`
#     ส่งเข้า PICT ตรง ๆ + Dart-side post-filter (กัน macOS PICT bug)

# ─── Format A: Dataset Constraints ───────────────────────────────────────────
job_01_title_textfield.valid   = วิศวกรซอฟต์แวร์อาวุโส
job_07_salary_min_textfield.valid = 30000
job_08_salary_max_textfield.valid = 40000

# ─── Format B: PICT IF/THEN Constraints ──────────────────────────────────────
# ถ้า category = Engineering → employment type ต้อง = Freelance (ตรงกับ DB)
# IF [search_02_category_dropdown] = "Engineering" THEN [search_03_type_dropdown] = "Freelance";
